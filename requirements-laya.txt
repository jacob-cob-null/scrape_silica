# Laya runtime for the Pax Silica AI decision filter.
# Install with: python -m pip install -r requirements-laya.txt
#
# IMPORTANT - transformers is pinned below 5:
#   transformers 5.x requires torch >= 2.6, but this machine runs
#   torch 2.5.1+cu121 (CUDA 12.1, RTX 3060). With transformers 5.x installed,
#   torch is silently disabled ("Disabling PyTorch because PyTorch >= 2.6 is
#   required") and model construction crashes with NameError: name 'nn' is not
#   defined. transformers 4.57.6 works with torch 2.5.1 and Laya 0.4.11.
#
# Side effect: this also keeps tokenizers at 0.22.2 and huggingface-hub at
# 0.36.2. If you ever upgrade torch to >= 2.6 you can relax the pin.
laya==0.4.11
transformers>=4.48,<5
